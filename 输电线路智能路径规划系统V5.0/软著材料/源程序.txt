def _select_final_route(methods, case_metrics):
    valid_methods = [
        method for method, path in methods.items()
        if path and method in case_metrics
        and case_metrics[method].get("hard_violations") == 0
    ]
    if "A*(baseline)" in valid_methods:
        return "A*(baseline)", None, "astar_feasible_priority"
    fallback = next((method for method in ("VIN-Grad", "PPO") if method in valid_methods), None)
    if fallback is not None:
        return fallback, None, "valid_candidate_fallback"
    return None, None, "no_hard_constraint_compliant_route"

def expand_single_cell_route(geo, start_ll, end_ll):
    """Turn a one-cell route into the short segment between distinct input endpoints."""
    if not geo or len(geo) != 1:
        return geo
    if tuple(start_ll) == tuple(end_ll):
        raise ValueError("起终点坐标相同，不能生成零长度线路")
    return [(start_ll[1], start_ll[0]), (end_ll[1], end_ll[0])]


def validate_route_hard_constraints(geo, transform, hard_mask):
    """Reject a candidate whose exported geometry crosses a blocked grid cell."""
    if not geo:
        return None, {"hard_constraint_status": "no_path", "hard_violations": 0}
    violations = path_hard_violations(geo, transform, hard_mask)
    report = {
        "hard_constraint_status": "pass" if violations == 0 else "fail",
        "hard_violations": int(violations),
    }
    return (geo if violations == 0 else None), report


def screening_data_quality(osm_layer_metadata):
    """Describe missing settlement proxy layers without treating them as empty land."""
    required_layers = ("buildings", "landuse")
    missing = [
        name for name in required_layers
        if osm_layer_metadata.get(name, {}).get("status") != "available"
    ]
    return ("incomplete_osm_proxies" if missing else "osm_proxy_features_observed"), missing


def validate_endpoint_cells(start, end, shape, hard_mask):
    """Require fixed endpoints to lie inside the DEM and the complete passable mask."""
    height, width = shape
    for label, (row, col) in (("起点", start), ("终点", end)):
        if not (0 <= row < height and 0 <= col < width):
            raise ValueError(f"{label}超出DEM范围")
        if not hard_mask[row, col]:
            raise ValueError(f"{label}位于NoData/建筑/配置禁区内；请核对接入点坐标或约束")
    return start, end


def run_method_or_skip_on_oom(method_name, run_method):
    """Skip a neural candidate only for PyTorch OOM; let other errors surface."""
    try:
        return run_method(), None
    except torch.OutOfMemoryError as exc:
        message = str(exc)
    print(f"  [{method_name}] 显存不足，跳过该方法并继续其余规划: {message}")
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
    return None, message

def load_case_reference(path, require_single_line=False):
    if not path or not Path(path).is_file():
        return None, "reference_missing"
    try:
        real_geo = load_real_path(
            path, require_single_line=require_single_line, raise_errors=True)
    except ValueError as exc:
        print(f"  [warn] 目标线路参考文件无效: {exc}")
        return None, "reference_invalid_or_empty"
    except Exception as exc:
        print(f"  [warn] 目标线路参考文件无法读取: {exc}")
        return None, "reference_invalid_or_unreadable"
    return (real_geo, "reference_loaded" if real_geo else "reference_invalid_or_empty")


def summarize_run_completion(expected_cases, results, failed_cases):
    failed_names = {item["case"] for item in failed_cases}
    no_route_cases = [
        record["case"] for record in results
        if record["metrics"].get(FINAL_ROUTE_LABEL, {}).get("planning_status")
        == "no_hard_constraint_compliant_route"
    ]
    no_route_names = set(no_route_cases)
    completed = {
        record["case"] for record in results
        if record["case"] not in failed_names and record["case"] not in no_route_names
    }
    return {
        "expected_cases": expected_cases,
        "completed_cases": len(completed),
        "recorded_cases": len(results),
        "failed_cases": failed_cases,
        "no_route_cases": no_route_cases,
        "complete": expected_cases > 0 and len(completed) == expected_cases
        and not failed_cases and not no_route_cases,
    }

def method_pass8_summary(results, method, expected_cases=None, failed_case_names=None):
    excluded = set(failed_case_names or ())
    eligible = [record for record in results
                if record["case"] not in excluded
                and record["metrics"].get(FINAL_ROUTE_LABEL, {}).get("planning_status")
                != "no_hard_constraint_compliant_route"]
    values = [record["metrics"].get(method, {}).get("pass8") for record in eligible]
    evaluated = [value for value in values if value is not None]
    errors = [record["metrics"].get(method, {}).get("worst_err_pct")
              for record in eligible
              if record["metrics"].get(method, {}).get("pass8") is not None
              and record["metrics"].get(method, {}).get("worst_err_pct") is not None]
    total_cases = max(len(results), expected_cases or 0)
    return {
        "pass": sum(value is True for value in evaluated),
        "fail": sum(value is False for value in evaluated),
        "unmeasured": total_cases - len(evaluated),
        "worst_err_pct": max(errors) if errors else None,
    }

def parse_lat_lon(value, label):
    try:
        parts = [float(part) for part in value.split(",")]
    except ValueError as exc:
        raise ValueError(f"{label}必须是纬度,经度数值") from exc
    if (len(parts) != 2 or not all(math.isfinite(part) for part in parts)
            or not -90 <= parts[0] <= 90 or not -180 <= parts[1] <= 180):
        raise ValueError(f"{label}经纬度无效；纬度须在[-90,90]、经度须在[-180,180]")
    return tuple(parts)

def cache_load_real_geo(arr):
    """兼容新旧缓存格式读取 real_geo, 统一为 折线列表, 每条 = [(lon,lat),...]。

    旧格式: np.array([real_geo], dtype=object) 可能落成 (1,1,N,2) 数值数组 或 (1,n) 对象数组;
    新格式: pickle 字节。
    """
    if arr.dtype == object:
        if arr.ndim == 1:
            val = arr[0]
        else:
            val = arr[0]  # (1,1,N,2) -> (1,N,2); (1,n) -> (n,)
    else:
        try:
            val = pickle.loads(arr.tobytes())
        except Exception:
            val = arr
    if val is None:
        return None
    if isinstance(val, np.ndarray):
        val = val.tolist()
    if isinstance(val, tuple):
        val = [val]
    out = []
    for poly in val:
        if isinstance(poly, np.ndarray):
            poly = poly.tolist()
        out.append([(float(x), float(y)) for x, y in poly])
    return out

_network_lock = threading.Lock()
_NETWORK_CACHE = {}  # path -> (coords_tuple_list, geom_list)


def load_network_lines(shp_path):
    """加载 SHP 中全部 LineString(将 MultiLineString 拆开), 进程级缓存。"""
    with _network_lock:
        if shp_path in _NETWORK_CACHE:
            return _NETWORK_CACHE[shp_path]
    import geopandas as gpd
    gdf = gpd.read_file(shp_path)
    coords_list, geom_list = [], []
    for geom in gdf.geometry:
        if geom is None or geom.is_empty:
            continue
        if geom.geom_type == "LineString":
            coords_list.append(tuple(geom.coords))
            geom_list.append(geom)
        elif geom.geom_type == "MultiLineString":
            for sub in geom.geoms:
                coords_list.append(tuple(sub.coords))
                geom_list.append(sub)
    with _network_lock:
        _NETWORK_CACHE[shp_path] = (coords_list, geom_list)
    return coords_list, geom_list

def _norm(c):
    return tuple((round(float(x), 6), round(float(y), 6)) for x, y in c)


def match_exclude_index(real_geo_coords, coords_list):
    """按坐标序列精确匹配待规划线在全网中的索引(留一排除); 支持反向。"""
    target = _norm(real_geo_coords)
    rev = tuple(reversed(target))
    for i, c in enumerate(coords_list):
        nc = _norm(c)
        if nc == target or nc == rev:
            return i
    return None

def regional_density_band(geom_list, exclude_idx, transform, H, W, sigma_px=12.0):
    """区域走廊密度: 栅格化既有线路(留一)后高斯平滑, 归一化到 [0,1]。

    sigma_px 建议 8~25 像元(对应约 1.5~5km @150m/px), 表达'区域走廊偏好'。
    """
    from rasterio.features import rasterize as rio_rasterize
    from shapely.geometry import mapping
    from scipy.ndimage import gaussian_filter

    geoms = [g for i, g in enumerate(geom_list) if i != exclude_idx] if exclude_idx is not None else geom_list
    mask = rio_rasterize([(mapping(g), 1) for g in geoms], out_shape=(H, W),
                         transform=transform, fill=0, dtype="float32", all_touched=False)
    dens = gaussian_filter(mask, sigma=sigma_px)
    mx = dens.max()

def corridor_orientation_band(geom_list, exclude_idx, transform, H, W, win=5):
    """每像元'最近既有线路段'的局部方向(弧度, 模π∈[0,π))。

    思路: 线像元上按窗口PCA取主轴方向, 再用距离变换把方向传播到全图。
    无线路像元(孤立区)返回 NaN。
    """
    from rasterio.features import rasterize as rio_rasterize
    from shapely.geometry import mapping
    from scipy.ndimage import distance_transform_edt
    from numpy.linalg import eig
    import math

    geoms = [g for i, g in enumerate(geom_list) if i != exclude_idx] if exclude_idx is not None else geom_list
    mask = rio_rasterize([(mapping(g), 1) for g in geoms], out_shape=(H, W),
                         transform=transform, fill=0, dtype="uint8", all_touched=False)
    orient = np.full((H, W), np.nan, np.float64)
    ys, xs = np.nonzero(mask)
    w = max(win, 3)
    for y, x in zip(ys, xs):
        y0, y1 = max(0, y - w), min(H, y + w + 1)
        x0, x1 = max(0, x - w), min(W, x + w + 1)
        sub = mask[y0:y1, x0:x1]
        pts = np.argwhere(sub > 0).astype(np.float64) + np.array([y0, x0])
        if len(pts) < 3:
            orient[y, x] = 0.0
            continue
        mu = pts.mean(axis=0)
        cov = (pts - mu).T @ (pts - mu) / len(pts)
        wv, v = eig(cov)
        ang = math.atan2(v[1, int(np.argmax(wv))], v[0, int(np.argmax(wv))])
        orient[y, x] = ang % math.pi
    _, idx = distance_transform_edt(1 - mask, return_indices=True)
    ir, ic = idx
    return orient[ir, ic]

def build_route_cost(terrain, osm, settlement_weight=2.0, model_cost=None):
    """Build an interpretable screening cost with a soft settlement proxy.

    The proxy is the maximum of normalized mapped-building density and mapped
    ``landuse=residential`` coverage. This combines complementary OSM signals
    without double-counting overlap. It is a soft preference only, not a hard
    exclusion or a substitute for population/building safety data.
    """
    slope = np.nan_to_num(terrain["slope"], nan=0.0, posinf=90.0, neginf=0.0)
    roughness = np.nan_to_num(terrain["roughness_27"], nan=0.0, posinf=0.0, neginf=0.0)
    terrain_cost = 0.75 * np.clip(slope / 45.0, 0.0, 1.0) ** 1.5
    terrain_cost += 0.25 * np.clip(roughness / 100.0, 0.0, 1.0)

    density = np.broadcast_to(np.asarray(
        osm.get("building_density", np.zeros_like(terrain_cost)), dtype=np.float32),
        terrain_cost.shape)
    density = np.nan_to_num(density, nan=0.0, posinf=0.0, neginf=0.0)
    positive = density[density > 0]
    if positive.size:
        p99 = max(float(np.percentile(positive, 99)), 1.0)
        building_proxy = np.clip(np.log1p(density) / np.log1p(p99), 0.0, 1.0)
    else:
        building_proxy = np.zeros_like(terrain_cost)
    residential = np.broadcast_to(np.asarray(
        osm.get("residential_landuse_mask", np.zeros_like(terrain_cost)), dtype=np.float32),
        terrain_cost.shape)
    residential_proxy = np.clip(np.nan_to_num(residential, nan=0.0, posinf=0.0, neginf=0.0), 0.0, 1.0)
    settlement_proxy = np.maximum(building_proxy, residential_proxy)

    if model_cost is not None:
        model = np.nan_to_num(model_cost, nan=0.0, posinf=1.0, neginf=0.0)
        terrain_cost = 0.5 * terrain_cost + 0.5 * np.clip(model, 0.0, 1.0)
    return (terrain_cost + max(float(settlement_weight), 0.0) * settlement_proxy).astype(np.float32)

def haversine_km(lon1, lat1, lon2, lat2):
    R = 6371.0
    p1, p2 = math.radians(lat1), math.radians(lat2)
    dp = math.radians(lat2 - lat1)
    dl = math.radians(lon2 - lon1)
    a = math.sin(dp / 2) ** 2 + math.cos(p1) * math.cos(p2) * math.sin(dl / 2) ** 2
    return R * 2 * math.atan2(math.sqrt(a), math.sqrt(1 - a))


def path_length_km(coords):
    return sum(haversine_km(coords[i][0], coords[i][1], coords[i + 1][0], coords[i + 1][1])
               for i in range(len(coords) - 1))


def point_to_polyline_km(lon, lat, polylines):
    """点到折线集合(多条折线)的最小 haversine 距离 km。polylines: [[(lon,lat),...],...]"""
    best = 1e9
    for poly in polylines:
        for i in range(len(poly) - 1):
            lon0, lat0 = poly[i]
            lon1, lat1 = poly[i + 1]
            d_m = math.hypot((lon1 - lon0) * 111000 * math.cos(math.radians((lat0 + lat1) / 2)),
                             (lat1 - lat0) * 111000)
            n = max(int(d_m / 30.0), 1)
            for k in range(n + 1):
                t = k / n
                plon = lon0 + (lon1 - lon0) * t
                plat = lat0 + (lat1 - lat0) * t
                d = haversine_km(lon, lat, plon, plat)
                if d < best:
                    best = d
    return best

    class VIN(nn.Module):
        """神经值传播网络：可微 Bellman 迭代生成代价-到此为止值图 V(s)。
        输入: cost(1) + goal(1) + extra(n_extra)；输出 V(1, H, W)。"""
        def __init__(self, n_extra=8, K=15, hidden=32):
            super().__init__()
            self.K = K
            self.enc = nn.Sequential(
                nn.Conv2d(3 + n_extra, hidden, 3, padding=1, bias=False),
                nn.BatchNorm2d(hidden), nn.ReLU(),
                nn.Conv2d(hidden, hidden, 3, padding=1, bias=False),
                nn.BatchNorm2d(hidden), nn.ReLU())
            # 消息传递栈：首层把 (s,V) 的 hidden+1 通道压回 hidden，
            # 后续 ResidualBlock 保持 hidden 通道，避免通道不匹配。
            self.trans = nn.Sequential(
                nn.Conv2d(hidden + 1, hidden, 3, padding=1, bias=False),
                nn.BatchNorm2d(hidden), nn.ReLU(),
                ResidualBlock(hidden),
                ResidualBlock(hidden))
            self.minpool = DifferentiableMinPool()
            self.value_head = nn.Sequential(
                nn.Conv2d(hidden, 32, 3, padding=1), nn.ReLU(),
                nn.Conv2d(32, 1, 1), nn.Softplus())

        def forward(self, cost, goal, extra, traversable, pixel_spacing):
            state = torch.cat([cost, goal, traversable, extra], 1)
            s = self.enc(state)
            V = torch.zeros_like(cost)
            for _ in range(self.K):
                target, _ = self.minpool(V, cost, traversable, pixel_spacing)
                x = torch.cat([s, V], 1)          # hidden + 1 通道
                x = self.trans(x)                 # -> hidden 通道
                delta = self.value_head(x)        # -> 1 通道 (非负)
                V = V + 0.1 * (target - V) + 0.05 * delta
                V = torch.where(goal > 0, torch.zeros_like(V), V)
                V = torch.where(traversable > 0.5, V, torch.zeros_like(V))
            return V

def _sample_grad(arr, r, c, eps=1.0):
    dy = (_bilin(arr, r + eps, c) - _bilin(arr, r - eps, c)) / (2 * eps)
    dx = (_bilin(arr, r, c + eps) - _bilin(arr, r, c - eps)) / (2 * eps)
    return dy, dx


def _bilin(arr, r, c):
    H, W = arr.shape
    r = min(max(r, 0), H - 1.001); c = min(max(c, 0), W - 1.001)
    r0, c0 = int(np.floor(r)), int(np.floor(c))
    r1, c1 = min(r0 + 1, H - 1), min(c0 + 1, W - 1)
    dr, dc = r - r0, c - c0
    return (arr[r0, c0] * (1 - dr) * (1 - dc) + arr[r0, c1] * (1 - dr) * dc +
            arr[r1, c0] * dr * (1 - dc) + arr[r1, c1] * dr * dc)

def smooth_cost(cost, sigma=1.0, mix=0.5):
    try:
        from scipy.ndimage import gaussian_filter
    except ImportError:
        return cost
    c = np.nan_to_num(np.asarray(cost, dtype=np.float32), nan=1.0, posinf=1.0, neginf=0.0)
    s = gaussian_filter(c, sigma=sigma).astype(np.float32)
    out = np.clip(mix * c + (1.0 - mix) * s, 0.0, 1.0).astype(np.float32)
    out[~np.isfinite(cost)] = np.nan
    return out

def compute_deviation(planned_geo, real_geo):
    """Project both routes to local meters and estimate symmetric Hausdorff at 5m spacing.
    """
    if not planned_geo or len(planned_geo) < 2 or not real_geo:
        return None
    try:
        from pyproj import CRS, Transformer
        from shapely import linestrings, points
        from shapely.strtree import STRtree

        real_polys = real_geo if isinstance(real_geo[0][0], (list, tuple)) else [real_geo]
        real_polys = [poly for poly in real_polys if len(poly) >= 2]
        if not real_polys:
            return None
        coords = list(planned_geo) + [point for poly in real_polys for point in poly]
        lons = np.radians([point[0] for point in coords])
        lon0 = math.degrees(math.atan2(float(np.sin(lons).mean()), float(np.cos(lons).mean())))
        lat0 = float(np.mean([point[1] for point in coords]))
        local_crs = CRS.from_proj4(
            f"+proj=aeqd +lat_0={lat0} +lon_0={lon0} +datum=WGS84 +units=m +no_defs"
        )
        project = Transformer.from_crs("EPSG:4326", local_crs, always_xy=True).transform

        def project_coordinates(poly):
            return np.asarray([project(lon, lat) for lon, lat in poly], dtype=np.float64)

        def densify(projected):
            samples = [projected[:1]]
            for start, end in zip(projected, projected[1:]):
                length = math.hypot(end[0] - start[0], end[1] - start[1])
                steps = max(1, int(math.ceil(length / 5.0)))
                fractions = np.arange(1, steps + 1, dtype=np.float64)[:, None] / steps
                samples.append(start + fractions * (end - start))
            return np.concatenate(samples, axis=0)

        planned_projected = project_coordinates(planned_geo)
        real_projected = [project_coordinates(poly) for poly in real_polys]
        planned_samples = densify(planned_projected)
        real_samples = [densify(poly) for poly in real_projected]

        def segment_tree(projected_lines):

def eval_error(plan_len_km, dev):
    """规划线 vs 真实线的 8% 误差度量。

    dev: compute_deviation 返回字典(含 real_length_km / dev_hausdorff_km, 单位 km)。
    返回:

      worst_err_pct: max(len_err_pct, haus_pct)
      pass8       : worst_err_pct <= ERROR_TOL_PCT

    if not dev:
        return None

    real_len = dev.get("real_length_km")
    if not real_len or real_len <= 0:
        return None
    plan_len = float(plan_len_km or 0.0)
    len_err = abs(plan_len - real_len) / real_len * 100.0
    haus = (float(dev.get("dev_hausdorff_km", 0.0)) / real_len) * 100.0  # 已为对称 Hausdorff
    haus_fwd = (float(dev.get("dev_haus_fwd_km", dev.get("dev_hausdorff_km", 0.0))) / real_len) * 100.0

        "haus_pct": round(haus, 6),

        "worst_err_pct": round(worst, 6),
        "pass8": bool(worst <= ERROR_TOL_PCT),
    }

def compute_distance_band(geoms, transform, shape, max_dist_m=5000.0):

def compute_building_density(geoms, transform, shape):
    on = rasterize_on(geoms, transform, shape)
    if on.sum() == 0:
        return np.zeros(shape, np.float32)
    cell_m = pixel_meters(transform, transform.f + transform.e * (shape[0] / 2.0))
    sigma = max(2, int(500 / cell_m))
    density = gaussian_filter(on.astype(np.float32), sigma=sigma)
    cell_area_km2 = (cell_m / 1000.0) ** 2
    return (density / cell_area_km2).astype(np.float32)

def _rdp(points, epsilon):
    """Ramer-Douglas-Peucker 折线简化。

    改用显式栈的迭代实现, 避免对长且高度弯曲的路径递归过深触发
    'maximum recursion depth exceeded' (Case 3 PPO 路径点数常达数千, 单边反复
    细分时递归深度可超 1000)。迭代版结果与递归版完全一致。
    """
    points = list(points)
    n = len(points)
    if n < 3:
        return points
    keep = [False] * n
    keep[0] = True
    keep[n - 1] = True
    stack = [(0, n - 1)]
    while stack:
        s_i, e_i = stack.pop()
        if e_i <= s_i + 1:
            continue
        s = np.array(points[s_i], np.float64)
        e = np.array(points[e_i], np.float64)
        lv = e - s
========== 第 61 页 / 合并源码行 3714-3774 / smoothing.py ==========
        ll = np.linalg.norm(lv)
        if ll < 1e-12:
            # 退化段: 端点已标记保留, 丢弃内部点(与原递归版行为一致)
            continue
        unit = lv / ll
        dmax, idx = 0.0, s_i + 1
        for i in range(s_i + 1, e_i):
            v = np.array(points[i], np.float64) - s
            proj = np.clip(np.dot(v, unit), 0, ll)
            closest = s + proj * unit
            d = np.linalg.norm(np.array(points[i], np.float64) - closest)
            if d > dmax:
                dmax, idx = d, i
        if dmax > epsilon:
            keep[idx] = True
            stack.append((s_i, idx))
            stack.append((idx, e_i))
    return [points[i] for i in range(n) if keep[i]]


def smooth_path(geo, spacing_m=90.0):
    if len(geo) < 3:
        return geo
    arr = np.array([(lon, lat) for lon, lat in geo], np.float64)
    eps_deg = spacing_m / 111000.0 * 1.5
    simplified = _rdp(arr.tolist(), eps_deg)
    if len(simplified) < 2:
        return geo
    simplified = np.array(simplified)
    xs, ys = simplified[:, 0], simplified[:, 1]
    seg = np.sqrt(np.diff(xs) ** 2 + np.diff(ys) ** 2)
    cum = np.concatenate([[0], np.cumsum(seg)])
    total = cum[-1]
    if total < 1e-9:
        return geo
    spacing_deg = spacing_m / 111000.0
    n = max(int(total / spacing_deg), len(simplified))
    sd = np.linspace(0, total, n)
    xi = np.interp(sd, cum, xs); yi = np.interp(sd, cum, ys)
    return [(float(xi[i]), float(yi[i])) for i in range(n)]

def densify_path_for_raster(coords, transform):

def _dict_reader(file_obj, label, required_fields):
    reader = csv.DictReader(file_obj)
    fields = reader.fieldnames or []
    if any(not field for field in fields) or len(fields) != len(set(fields)):
        raise ValueError(f"{label}表头存在空列名或重复列名")
    missing = required_fields - set(fields)
    if missing:
        raise ValueError(f"{label}缺少列: {', '.join(sorted(missing))}")
    return reader

def plot_algo_comparison(all_results, out_path):
    """多算法指标聚合柱状图。"""
    methods = ["VIN-Grad", "PPO", "A*(baseline)"]
    metrics_keys = ["length_km", "mean_cost", "sinuosity", "hard_violations", "time_s", "cum_relief_m"]
    agg = {m: {k: [] for k in metrics_keys} for m in methods}
    for rec in all_results:
        for m, met in rec["metrics"].items():
            if m not in agg:   # 跳过最终路线记录等非量化对比条目
                continue
            for k in metrics_keys:
                if k in met and met[k] is not None and not (isinstance(met[k], float) and math.isnan(met[k])):
                    agg[m][k].append(met[k])
    means = {m: {k: (sum(v) / len(v) if v else 0.0) for k, v in agg[m].items()} for m in methods}
    labels = {
        "length_km": "路径长度 (km)", "mean_cost": "平均建设代价",
        "sinuosity": "弯曲度", "hard_violations": "硬约束违规(点)",
        "time_s": "耗时 (s)", "cum_relief_m": "累计高差 (m)"}
    fig, axes = plt.subplots(2, 3, figsize=(15, 9), dpi=140)
    axes = axes.ravel()
    colors = {"VIN-Grad": "#1f77ff", "PPO": "#2ca02c", "A*(baseline)": "#ff7f0e"}
    for i, k in enumerate(metrics_keys):
        ax = axes[i]
        vals = [means[m][k] for m in methods]
        bars = ax.bar(methods, vals, color=[colors[m] for m in methods])
        ax.set_title(labels[k])
        ax.set_ylabel(labels[k])
        for b, v in zip(bars, vals):
            ax.text(b.get_x() + b.get_width() / 2, v, f"{v:.2f}", ha="center", va="bottom", fontsize=9)
        ax.tick_params(axis="x", rotation=15)
    fig.suptitle("多算法路径规划量化对比（各走廊均值）", fontsize=14)
    fig.tight_layout(rect=[0, 0, 1, 0.96])
    try:
        fig.savefig(out_path, dpi=140)
    except Exception:
        import time as _time
        try:
            _time.sleep(0.2)
            _alt = out_path.with_name(out_path.stem + "_1" + out_path.suffix)
            fig.savefig(_alt, dpi=140)
            out_path = _alt
        except Exception as _se:
            plt.close(fig)
            print(f"  [WARN] 指标对比图保存失败(已跳过): {_se}")
            return
    plt.close(fig)
    print(f"  指标对比图保存: {out_path}")
